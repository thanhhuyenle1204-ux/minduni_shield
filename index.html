// Service Worker background.js – Tự động đồng bộ tin tuyển sinh chính thống mỗi 24 giờ
chrome.runtime.onInstalled.addListener(() => {
  chrome.alarms.create("minduniDailyAdmissionsSync", { periodInMinutes: 1440 });
  syncDailyOfficialAdmissions();
});

chrome.alarms.onAlarm.addListener((alarm) => {
  if (alarm.name === "minduniDailyAdmissionsSync") {
    syncDailyOfficialAdmissions();
  }
});

async function syncDailyOfficialAdmissions() {
  const todayStr = new Date().toLocaleDateString("vi-VN");
  const latestFeed = [
    `[${todayStr}] Bộ GD&ĐT giữ ổn định cấu trúc đề 2+2 bám sát chương trình lớp 12.`,
    `[${todayStr}] Hệ thống lọc ảo tự động xét phương thức có lợi nhất từ NV1 xuống dưới.`
  ];
  await chrome.storage.local.set({ minduniDailyNews: latestFeed, lastSync: todayStr });
}
